# Voiceger Zundamon TTS -- Python requirements.
#
# IMPORTANT -- environment isolation:
#   This repo is developed under direnv + Nix, which exports a huge
#   PYTHONPATH pointing at Nix site-packages for a different Python version.
#   If that leaks into this venv, imports fail with "Version mismatch"
#   cffi/numpy errors.  The launcher (run-voiceger-server.sh) unsets
#   PYTHONPATH and friends before starting the server; mirror that in any
#   ad-hoc invocation.
#
# Default engine (VOICEGER_ENGINE=audio8): Audio8/Audio8-TTS-Preview-0.1b,
# loaded through plain `transformers` with trust_remote_code=True.
#
# torch: platform-specific wheels are installed first by setup-voiceger.sh
# (cu121 index on Linux, stock PyPI wheels on macOS -- there is no CUDA
# build for Darwin). Pinned here too so resolving this file cannot silently
# drag in an incompatible torch:
torch>=2.5.0
torchaudio>=2.5.0

transformers>=4.57.0,<5
soundfile>=0.12
safetensors>=0.4
# transformers.audio_utils hard-imports librosa (used by AutoProcessor here)
# even though transformers does not declare it as an install dependency --
# leave unpinned so it pulls a modern release (needs importlib.metadata, not
# the pkg_resources API that current setuptools no longer ships).
librosa

fastapi
uvicorn
pydantic>=2
